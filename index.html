}; // ========================================== // 2. Canvas 各種形狀繪製邏輯 // ========================================== function drawStone(ctx, x, y, radius, stoneConfig, time = 0) { ctx.save(); ctx.translate(x, y); // 1. 發光特效（如有設定 pulse 則有呼吸燈效果） if (stoneConfig.glow) { let glowAlpha = 1; if (stoneConfig.pulse) { glowAlpha = (Math.sin(time / 300) + 1) / 2 * 0.5 + 0.5; // 0.5 ~ 1.0 呼吸變化 } ctx.shadowColor = stoneConfig.glow; ctx.shadowBlur = 15 * glowAlpha; } ctx.fillStyle = stoneConfig.color; ctx.beginPath(); // 2. 依據 shape 繪製幾何線條 switch (stoneConfig.shape) { case 'pebble': // 平滑鵝卵石/圓形 ctx.arc(0, 0, radius, 0, Math.PI * 2); break; case 'crystal': // 尖銳水晶柱 ctx.moveTo(0, -radius * 1.3); ctx.lineTo(radius * 0.7, -radius * 0.4); ctx.lineTo(radius * 0.6, radius); ctx.lineTo(-radius * 0.6, radius); ctx.lineTo(-radius * 0.7, -radius * 0.4); break; case 'octahedron': // 雙錐八面體 (夜光石) ctx.moveTo(0, -radius * 1.2); ctx.lineTo(radius * 0.8, 0); ctx.lineTo(0, radius * 1.2); ctx.lineTo(-radius * 0.8, 0); break; case 'nugget': // 不規則狗頭金塊 const pts = [ [0, -1], [0.8, -0.7], [1.1, 0.1], [0.6, 0.9], [-0.4, 1.0], [-1.0, 0.3], [-0.7, -0.8] ]; pts.forEach(([px, py], i) => { const rx = px * radius, ry = py * radius; if (i === 0) ctx.moveTo(rx, ry); else ctx.lineTo(rx, ry); }); break; case 'star': // 五角星 for (let i = 0; i < 10; i++) { const r = (i % 2 === 0) ? radius * 1.1 : radius * 0.5; const angle = (i * Math.PI) / 5 - Math.PI / 2; ctx.lineTo(Math.cos(angle) * r, Math.sin(angle) * r); } break; case 'teardrop': // 水滴 / 勾玉形 ctx.moveTo(0, -radius * 1.2); ctx.bezierCurveTo(radius * 1.2, -radius * 0.2, radius * 0.8, radius, 0, radius); ctx.bezierCurveTo(-radius * 0.8, radius, -radius * 1.2, -radius * 0.2, 0, -radius * 1.2); break; case 'snowflake': // 六角雪花 for (let i = 0; i < 6; i++) { const angle = (i * Math.PI) / 3; ctx.moveTo(0, 0); ctx.lineTo(Math.cos(angle) * radius, Math.sin(angle) * radius); } ctx.strokeStyle = stoneConfig.color; ctx.lineWidth = 3; ctx.stroke(); ctx.restore(); return; case 'heptagon': // 正七邊形 default: for (let i = 0; i < 7; i++) { const angle = (i * 2 * Math.PI) / 7; ctx.lineTo(Math.cos(angle) * radius, Math.sin(angle) * radius); } break; } ctx.closePath(); ctx.fill(); // 高光細節 (增添立體感) ctx.fillStyle = 'rgba(255, 255, 255, 0.25)'; ctx.beginPath(); ctx.arc(-radius * 0.2, -radius * 0.3, radius * 0.3, 0, Math.PI * 2); ctx.fill(); ctx.restore(); } // ========================================== // 3. 成就判定邏輯 // ========================================== function checkAndGrantAchievements(userProgress, checkInDate = new Date()) { const newAchievements = []; const hours = checkInDate.getHours(); const month = checkInDate.getMonth() + 1; const date = checkInDate.getDate(); // 1. 連續簽到 7 天 if (userProgress.streakDays >= 7 && !userProgress.unlockedAchievements.includes('ACH_7_DAYS')) { newAchievements.push({ id: 'ACH_7_DAYS', rewardStone: STONES.CHRONO_METEOR }); } // 2. 累計簽到 30 天 if (userProgress.totalDays >= 30 && !userProgress.unlockedAchievements.includes('ACH_30_DAYS')) { newAchievements.push({ id: 'ACH_30_DAYS', rewardStone: STONES.GOLD_NUGGET }); } // 3. 深夜簽到 (23:00~02:00) 滿 5 次 const isNight = (hours >= 23 || hours < 2); if (isNight) { userProgress.nightCount = (userProgress.nightCount || 0) + 1; if (userProgress.nightCount >= 5 && !userProgress.unlockedAchievements.includes('ACH_NIGHT_5')) { newAchievements.push({ id: 'ACH_NIGHT_5', rewardStone: STONES.FLUORITE }); } } // 4. 特定日期簽到：1/1 元旦 if (month === 1 && date === 1 && !userProgress.unlockedAchievements.includes('ACH_NEW_YEAR')) { newAchievements.push({ id: 'ACH_NEW_YEAR', rewardStone: STONES.NEW_YEAR_STAR }); } // 5. 特定日期簽到：12/25 聖誕節 if (month === 12 && date === 25 && !userProgress.unlockedAchievements.includes('ACH_XMAS')) { newAchievements.push({ id: 'ACH_XMAS', rewardStone: STONES.XMAS_CRYSTAL }); } return { newAchievements, isNight }; } // ========================================== // 4. 簽到抽獎與石頭獲得邏輯 // ========================================== function getDailyStone(userProgress, checkInDate = new Date()) { const month = checkInDate.getMonth() + 1; const date = checkInDate.getDate(); const hours = checkInDate.getHours(); const isNight = (hours >= 23 || hours < 2); // A. 優先規則：特定節日當天簽到 100% 必得節日石 if (month === 1 && date === 1) return STONES.NEW_YEAR_STAR; if (month === 12 && date === 25) return STONES.XMAS_CRYSTAL; // B. 幸運觸發：無條件彩虹蛋白石 (3%) if (Math.random() < 0.03) return STONES.OPAL; // C. 深夜時段條件觸發 if (isNight) { // 已解鎖夜貓子成就，25% 獲得夜光螢石 if (userProgress.unlockedAchievements.includes('ACH_NIGHT_5') && Math.random() < 0.25) { return STONES.FLUORITE; } // 深夜 50% 機率獲得黑玻璃隕石 if (Math.random() < 0.50) { return STONES.METEORITE; } } // D. 依已解鎖成就擴充動態掉落池 let dynamicPool = [ STONES.RIVER_PEBBLE, STONES.ROSE_QUARTZ, STONES.AMETHYST ]; // 若已解鎖【積少成多】(30天)，5% 機率掉落黃金狗頭金 if (userProgress.unlockedAchievements.includes('ACH_30_DAYS') && Math.random() < 0.05) { return STONES.GOLD_NUGGET; } // 若已解鎖【持之以恆】(7天)，8% 機率掉落時空隕石 if (userProgress.unlockedAchievements.includes('ACH_7_DAYS') && Math.random() < 0.08) { return STONES.CHRONO_METEOR; } // E. 預設：從基礎池隨機抽取 const randomIndex = Math.floor(Math.random() * dynamicPool.length); return dynamicPool[randomIndex]; }
